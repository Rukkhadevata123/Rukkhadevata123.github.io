    提示

    建议参考

    切换 x86_64-v3 仓库 #

    bash
    echo 'APT::Architecture-Variants "amd64v3";' | sudo tee /etc/apt/apt.conf.d/99enable-amd64v3
sudo apt update && sudo apt upgrade

    WSL 互操作 #

    bash
    sudo sh -c 'echo :WSLInterop:M::MZ::/init:PF > /usr/lib/binfmt.d/WSLInterop.conf'
sudo systemctl restart systemd-binfmt
code .

    debootstrap #

    bash
    dpkg-reconfigure tzdata
dpkg-reconfigure locales
sudo apt install linux-generic grub-efi-amd64-signed shim-signed

    调大字体 #

    微信依赖 #

    bash
    sudo apt install libxcb-icccm4 libxcb-image0 libxcb-render-util0 libxcb-keysyms1

    输入法 #

    bash
    cd
git clone https://github.com/rime/plum.git plum
cd plum
bash rime-install iDvel/rime-ice

    外接显示器 #

    bash
    sudo apt install ddcutil i2c-tools
sudo ddcutil detect
sudo ddcutil setvcp 10 70 --display 1

    截图标注 #

    bash
    sudo snap install gradia

    然后添加快捷键

    bash
    gradia --screenshot=INTERACTIVE

    双系统时间 #

    bash
    sudo timedatectl set-local-rtc 0

    windows 注意

    cmd
    reg add "HKEY_LOCAL_MACHINE\System\CurrentControlSet\Control\TimeZoneInformation" /v RealTimeIsUniversal /d 1 /t REG_DWORD /f

    启用 root 账户 #

    bash
    sudo passwd

    开启休眠 #

    往 /etc/dracut.conf.d/resume.conf 写入

    bash
    add_dracutmodules+=" resume "
hostonly="no"

    然后 sudo update-initramfs

    安装解码器和微软字体 #

    bash
    sudo apt install ubuntu-restricted-addons intel-media-va-driver-non-free nvidia-vaapi-driver -y

    安装 Nvidia 显卡驱动 #

    bash
    sudo ubuntu-drivers install

    开启 zswap 以及使用 xe 驱动 #

    bash
    lspci -nnd ::03xx

    然后内核参数

    bash
    GRUB_CMDLINE_LINUX_DEFAULT="quiet zswap.enabled=1 zswap.compressor=zstd i915.force_probe=!a7a0 xe.force_probe=a7a0"

    干净卸载 #

    bash
    sudo apt autoremove --purge <package name>

    配置代理 #

    设置 Qt 主题 #

    bash
    sudo apt install qt6-style-kvantum qt6ct
echo 'export QT_STYLE_OVERRIDE=qt6ct' | sudo tee /etc/profile.d/qt.sh

    美化终端 #

    参见https://github.com/microsoft/WindowsDeveloperConfig

    虚拟机 #

    bash
    sudo apt install virt-manager

    安装vscode #

    bash
    wget "https://code.visualstudio.com/sha/download?build=stable&os=linux-deb-x64" -O code.deb
mv code.deb /tmp
cd /tmp
sudo apt install ./code.deb

    HiDPI #

    • Qt应用程序其他选项QT_AUTO_SCREEN_SCALE_FACTOR=1 QT_SCALE_FACTOR=1.5
    • Java应用程序缩放问题java -jar -Dglass.gtk.uiScale=150% xxx.jar

    文件极速查找工具 #

    bash
    sudo apt install plocate
sudo updatedb
# sudo updatedb -U /path
locate something

    更多软件包管理器前端 #

    bash
    sudo apt install synaptic aptitude

    索引文件所属包 #

    bash
    sudo apt install apt-file
sudo apt-file update

    更新系统 #

    bash
    echo 'alias update="sudo apt update && sudo apt upgrade && sudo apt autoremove --purge && sudo apt autoclean && sudo snap refresh"' >> ~/.zshrc

    移除旧版本Snap包 #

    bash
    LANG=en_US.UTF-8; snap list --all | awk '/disabled/ { system("sudo snap remove " $1 " --revision=" $3) }'

    磁盘空间查看器 #

    bash
    sudo apt install baobab

    修复NTFS磁盘无法挂载 #

    bash
    sudo lsblk
sudo ntfsfix --clear-dirty /dev/nvme0n1p4

    把桌面换成KDE #

    bash
    sudo apt install kubuntu-desktop

    然后自行卸载安装输入法

    增大vm.max_map_count以提升游戏体验 #

    bash
    echo 'vm.max_map_count = 2147483642' | sudo tee /etc/sysctl.d/80-gamecompatibility.conf

    matlab内存不足 #

    bash
    echo "* hard nofile 65536" | sudo tee -a /etc/security/limits.d/systemd.conf
